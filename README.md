# ==============================================================================
# 4. 대본 분석 및 최적 촬영지 추천 엔진 함수
# ==============================================================================
def recommend_locations(script_text, region_filter="전체", top_n=3):
    # (1) 입력된 대본을 TF-IDF 벡터로 변환 후 4대 무드 확률 예측
    vec = vectorizer.transform([script_text])
    probs = model.predict_proba(vec)[0]
    
    # ['온기', '긴장', '고독', '몽환'] 순서대로 확률 정리
    class_prob_dict = dict(zip(model.classes_, probs))
    raw_mood_vec = np.array([class_prob_dict[m] for m in mood_labels])
    
    # 대본의 주된 무드 특성이 잘 드러나도록 제곱 정규화(Sharpening) 적용
    sharpened_vec = (raw_mood_vec ** 2)
    sharpened_vec = (sharpened_vec / sharpened_vec.sum()).reshape(1, -1)
    
    # (2) 희망 권역 필터링
    filtered_df = df_loc.copy()
    if region_filter != "전체":
        filtered_df = filtered_df[filtered_df['권역'] == region_filter].copy()
        if filtered_df.empty:
            print(f"[안내] '{region_filter}' 권역 데이터가 없어 전체 권역에서 검색합니다.")
            filtered_df = df_loc.copy()
            
    # (3) 대본 무드 벡터와 장소 무드 벡터 간 코사인 유사도 계산
    loc_mood_matrix = filtered_df[mood_labels].values
    similarities = cosine_similarity(sharpened_vec, loc_mood_matrix)[0]
    
    # 최종 추천 점수 = 무드 일치도(80%) + 촬영 편의성 지수(20%)
    filtered_df['무드일치도(%)'] = np.round(similarities * 100, 1)
    filtered_df['최종적합도'] = np.round((similarities * 100 * 0.8) + (filtered_df['촬영편의성'] * 0.2), 1)
    
    # (4) 분석 결과 리포트 출력
    print("\n" + "=" * 68)
    print(f"🎬 [입력 대본 / 연출 의도]\n\"{script_text}\"")
    print("-" * 68)
    print("📊 [AI 무드 분석 결과 (TF-IDF + 로지스틱 회귀)]")
    sorted_moods = sorted(class_prob_dict.items(), key=lambda x: x[1], reverse=True)
    for m, p in sorted_moods:
        bar = "■" * int(p * 25)
        print(f"  - {m}: {p*100:5.1f}%  {bar}")
    print(f"  👉 최종 분류 무드: [{sorted_moods[0][0]}]")
    print("-" * 68)
    print(f"📍 [추천 촬영지 TOP {top_n}] (희망 권역: {region_filter})")
    
    top_locs = filtered_df.sort_values(by='최종적합도', ascending=False).head(top_n)
    for rank, (_, row) in enumerate(top_locs.iterrows(), 1):
        print(f"  {rank}위. {row['장소명']} ({row['권역']} | {row['공간유형']})")
        print(f"      · 최종 적합도: {row['최종적합도']}점 (무드 일치도 {row['무드일치도(%)']}% / 촬영 편의성 {row['촬영편의성']}점)")
        print(f"      · 장소 무드 프로파일: 온기 {row['온기']} | 긴장 {row['긴장']} | 고독 {row['고독']} | 몽환 {row['몽환']}")
        print(f"      · 추천 근거: {row['특징']}\n")
